from flask import Flask, render_template, request
import time
app = Flask(__name__)
# ================= HOME =================
@app.route('/')
def home():
return render_template('index.html', welcome=True)
# ================= DIFFIE HELLMAN =================
@app.route('/diffie', methods=['GET', 'POST'])
def diffie():
result = None
welcome = True
if request.method == 'POST':
welcome = False
time.sleep(2)
p = int(request.form['p'])
g = int(request.form['g'])
a = int(request.form['a'])
b = int(request.form['b'])
# Public keys
A = pow(g, a, p)
B = pow(g, b, p)
# Shared keys
key1 = pow(B, a, p)
key2 = pow(A, b, p)
result = {
"p": p,
"g": g,
"a": a,
"b": b,
"A": A,
"B": B,
"key1": key1,
"key2": key2
}
return render_template('diffie.html', result=result, welcome=welcome)
if __name__ == '__main__':
app.run(debug=True)
