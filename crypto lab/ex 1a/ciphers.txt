# ================= SHIFT =================
def shift_encrypt(text, key):
result = ""
steps = ""
for i, ch in enumerate(text):
if ch.isalpha():
base = 'a' if ch.islower() else 'A'
original = ord(ch) - ord(base)
shifted = (original + key) % 26
new_char = chr(shifted + ord(base))
result += new_char
steps += f"Step {i+1}: {ch} -> ({original}+{key}) mod 26 =
{new_char}\n"
else:
result += ch
return result, steps
def shift_decrypt(text, key):
result = ""
steps = ""
for i, ch in enumerate(text):
if ch.isalpha():
base = 'a' if ch.islower() else 'A'
original = ord(ch) - ord(base)
shifted = (original - key) % 26
new_char = chr(shifted + ord(base))
result += new_char
steps += f"Step {i+1}: {ch} -> ({original}-{key}) mod 26 =
{new_char}\n"
else:
result += ch
return result, steps
# ================= VIGENERE =================
def vigenere_encrypt(text, key):
result = ""
steps = ""
j = 0
key = key.lower()
for i, ch in enumerate(text):
if ch.isalpha():
base = 'a' if ch.islower() else 'A'
p = ord(ch) - ord(base)
k = ord(key[j % len(key)]) - ord('a')
c = (p + k) % 26
new_char = chr(c + ord(base))
result += new_char
steps += f"Step {j+1}: ({ch}+{key[j%len(key)]}) -> {new_char}\n"
j += 1
else:
result += ch
return result, steps
def vigenere_decrypt(text, key):
result = ""
steps = ""
j = 0
key = key.lower()
for i, ch in enumerate(text):
if ch.isalpha():
base = 'a' if ch.islower() else 'A'
c = ord(ch) - ord(base)
k = ord(key[j % len(key)]) - ord('a')
p = (c - k) % 26
new_char = chr(p + ord(base))
result += new_char
steps += f"Step {j+1}: ({ch}-{key[j%len(key)]}) -> {new_char}\n"
j += 1
else:
result += ch
return result, steps
# ================= HILL =================
def hill_encrypt(text, key):
text = ''.join(filter(str.isalpha, text)).upper()
if len(text) % 2 != 0:
text += 'X'
k = list(map(int, key.split()))
if len(k) != 4:
return "", "Key must have 4 numbers"
result = ""
steps = ""
for i in range(0, len(text), 2):
p1 = ord(text[i]) - 65
p2 = ord(text[i+1]) - 65
c1 = (k[0]*p1 + k[1]*p2) % 26
c2 = (k[2]*p1 + k[3]*p2) % 26
result += chr(c1 + 65) + chr(c2 + 65)
steps += f"Block {i//2+1}: [{text[i]} {text[i+1]}] -> [{result[-2]}
{result[-1]}]\n"
return result, steps
def hill_decrypt(text, key):
k = list(map(int, key.split()))
if len(k) != 4:
return "", "Key must have 4 numbers"
det = (k[0]*k[3] - k[1]*k[2]) % 26
# find modular inverse
inv = -1
for i in range(26):
if (det * i) % 26 == 1:
inv = i
if inv == -1:
return "", "Key not invertible"
inv_key = [
( k[3]*inv) % 26,
(-k[1]*inv) % 26,
(-k[2]*inv) % 26,
( k[0]*inv) % 26
]
result = ""
steps = ""
for i in range(0, len(text), 2):
c1 = ord(text[i]) - 65
c2 = ord(text[i+1]) - 65
p1 = (inv_key[0]*c1 + inv_key[1]*c2) % 26
p2 = (inv_key[2]*c1 + inv_key[3]*c2) % 26
result += chr(p1 + 65) + chr(p2 + 65)
steps += f"Block {i//2+1}: [{text[i]} {text[i+1]}] -> [{result[-2]}
{result[-1]}]\n"
return result, steps
# ================= ROUTES =================
@app.route("/<cipher>/<mode>", methods=["POST"])
def run(cipher, mode):
data = request.json
text = data["text"]
key = data["key"]
if cipher == "shift":
key = int(key)
func = shift_encrypt if mode == "encrypt" else shift_decrypt
elif cipher == "vigenere":
func = vigenere_encrypt if mode == "encrypt" else vigenere_decrypt
elif cipher == "hill":
func = hill_encrypt if mode == "encrypt" else hill_decrypt
else:
return jsonify({"result": "", "steps": "Invalid cipher"})
result, steps = func(text, key)
return jsonify({"result": result, "steps": steps})
if __name__ == "__main__":
app.run(debug=True)
